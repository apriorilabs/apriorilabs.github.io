#include <priori/priori.hpp>
#include <priori/priori_dsp.hpp>
#include <priori/priori_lab.hpp>

int main(int argc, char** argv) {
    priori::AppConfig cfg{
        .title          = "TACTICAL SENSOR CONSOLE",
        .classification = PRIORI_CLASS_UNCLASSIFIED,  // Engine-enforced DoDI 5200.48 banners (unclippable)
        .env_style      = PRIORI_ENV_COASTAL_RADAR    // 64-bit WGS84 terrain (VK 1.4 RT + VK 1.2 Non-RT)
    };
    priori::App app(cfg, argc, argv);                 // Offscreen CI & LLM loop (--headless-capture / --json)

    // 1. Shared VRAM Tensors (C++/Slang or external CUDA/DMA-BUF writes -> GPU plots; 0 localhost IPC)
    auto tIq   = app.create_tensor_1d("IQ_RAW",  512);
    auto tPsd  = app.create_tensor_1d("PSD_DB",  256);
    auto tWf   = app.create_tensor_2d("SPEC_WF", 256, 256);
    float linkedCursorU = 0.5f;

    priori::dsp::DspPipeline dsp;
    dsp.add_node("RADIX2_FFT", "IQ_RAW", "PSD_DB", [&]() {
        priori::dsp::compute_iq_psd_db(tIq.data(), 256u, tPsd.data(), /*fft_shift=*/true);
    });
    priori::lab::Inspector lab;                       // Integrated DAG profiler, wire-probe & SHA-256 T&E DVR

    // 2. Deterministic VSync Frame Loop (Peak-hold preserves sub-ms transients between display refreshes)
    app.run([&](priori::FrameContext& ctx) {
        dsp.execute_all();
        tWf.accumulate_peak_hold_in_current_row(tPsd.data());
        if (ctx.step_waterfalls) tWf.push_row(tPsd.data());

        ctx.world.add_track({101u, 1200.0f, 240.0f, 1500.0f, 90.0f, 38.0f,
                             PRIORI_AFFIL_HOSTILE, PRIORI_DIM_AIR, priori::Palette::Coral,
                             4000.0f, 4.0f, "UAS-201", "S-BAND EMITTER"});

        const auto nav = ctx.ui.begin_header_bar(cfg.title, "ZERO IPC MIDDLEWARE", ctx.gpu_frame_us, ctx.fps);
        lab.draw(ctx, {nav.x + 16.0f, nav.y, 188.0f, nav.h}, dsp);

        auto card = ctx.ui.begin_card({16.0f, ctx.ui.top_safe_y() + 58.0f, 400.0f, 410.0f}, "SPECTRUM & WATERFALL");
        ctx.ui.plot_line_1d(card.next_row(85.0f), tPsd, 0.0f, 55.0f, priori::Palette::Ice, &linkedCursorU);
        ctx.ui.plot_waterfall_2d(card.next_row(230.0f), tWf, 4.0f, 52.0f, PRIORI_CMAP_INFERNO_EW, &linkedCursorU);
    });
}
